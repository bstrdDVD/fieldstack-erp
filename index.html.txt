function bdg(e){
  const M={activa:['b-v','ACTIVA'],finalizada:['b-g','FIN'],borrador:['b-g','BORRADOR'],

    entregado_completo:['b-v','COMPLETO'],confirmado:['b-v','CONFIRMADO'],
    urgente:['b-r','URGENTE'],normal:['b-b','NORMAL']};
  const[c,l]=M[e]||['b-g',e.toUpperCase()];
  return `<span class="badge ${c}">${l}</span>`;
}

.flatMap(pm=>(pm.items||[]).filter(i=>i.guia&&i.saldo>0)

}

function renderVCDists(){
const qs=C.quiebres[OID]||[];
return VC_S.dists.map((d,i)=>` <div class="qrow" style="margin-bottom:6px"> ${VC_S.dists.length>1?`<button class="qrow-del" onclick="rmVCD(${i})">✕</button>`:''} <div class="fr3"> <div class="fg" style="margin:0"><label class="fl">Quiebre destino</label> <select class="fs" id="vd-q-${i}" onchange="VC_S.dists[${i}].quiebre=this.value;updVCS()"> <option value="">Seleccioná...</option> ${qs.map(q=>`<option value=”${q.codigo}” ${d.quiebre===q.codigo?‘selected’:’’}>${q.codigo} — ${q.desc}</option>`).join('')} ${qs.flatMap(q=>(q.items_adicionales||[]).map(ex=>`<option value=”${q.codigo}/${ex.codigo}” ${d.quiebre===q.codigo+’/’+ex.codigo?‘selected’:’’}>${q.codigo}/${ex.codigo} [adicional]</option>`)).join('')} </select></div> <div class="fg" style="margin:0"><label class="fl">Cantidad (${VC_S.unidad})</label> <input class="fi" id="vd-c-${i}" type="number" placeholder="0" value="${d.cantidad}" oninput="VC_S.dists[${i}].cantidad=this.value;updVCS()"/></div> <div class="fg" style="margin:0"><label class="fl">Observación</label> <input class="fi" id="vd-o-${i}" placeholder="Opcional" value="${d.obs||''}" oninput="VC_S.dists[${i}].obs=this.value"/></div> </div> </div>`).join(’’);
}

function onVCG(val){
const b=$(‘vcbody’);
if(!val){b.style.display=‘none’;return;}
const[guia,mc]=val.split(’|’);
const pm=C.pms.find(p=>p.obra_id===OID&&(p.items||[]).some(i=>i.guia===guia&&i.mat_cod===mc));
const it=pm?.items?.find(i=>i.guia===guia&&i.mat_cod===mc);
if(!it){b.style.display=‘none’;return;}
VC_S={guia,mat_cod:mc,saldo:it.saldo||0,unidad:it.unidad||‘u’,dists:[{quiebre:’’,cantidad:’’,obs:’’}]};
b.style.display=‘block’;updVCS();
$(‘vcdists’).innerHTML=renderVCDists();
}

function updVCS(){
const tot=VC_S.dists.reduce((a,d)=>a+(parseFloat(d.cantidad)||0),0);
const rest=VC_S.saldo-tot;
const ok=tot>0&&rest>=0;
$(‘vcsaldo’).innerHTML=`Saldo guía <strong>${VC_S.guia}</strong>: <strong>${VC_S.saldo} ${VC_S.unidad}</strong> · Distribuido: <strong>${tot} ${VC_S.unidad}</strong> · Restante: <strong style="color:${rest<0?'var(--err)':rest===0?'var(--volt)':'var(--warn)'}">${rest} ${VC_S.unidad}</strong>`;
const btn=$(‘vcbtn’);if(btn)btn.disabled=!ok;
}

function addVCD(){VC_S.dists.push({quiebre:’’,cantidad:’’,obs:’’});$(‘vcdists’).innerHTML=renderVCDists();updVCS();}
function rmVCD(i){VC_S.dists.splice(i,1);$(‘vcdists’).innerHTML=renderVCDists();updVCS();}

function gVC(){
const o=C.obras.find(x=>x.id===OID);
const dists=VC_S.dists.map((_,i)=>{
const qSel=$(`vd-q-${i}`);
const txt=qSel?.options[qSel?.selectedIndex]?.text||’’;
return{quiebre:$(`vd-q-${i}`)?.value||VC_S.dists[i].quiebre,
desc:txt,cantidad:parseFloat($(`vd-c-${i}`)?.value||0)||0,obs:$(`vd-o-${i}`)?.value||’’};
}).filter(d=>d.quiebre&&d.cantidad>0);
if(!dists.length){toast(‘Completá al menos una distribución’,‘r’);return;}
const tot=dists.reduce((a,d)=>a+d.cantidad,0);
const vcId=`VC-${o.id}-${String(Date.now()).slice(-4)}`;
// Descontar saldo local
C.pms.forEach(pm=>{if(pm.obra_id===OID)(pm.items||[]).forEach(it=>{if(it.guia===VC_S.guia&&it.mat_cod===VC_S.mat_cod)it.saldo=Math.max(0,(it.saldo||0)-tot);});});
const inv=C.inv[OID]||[];const inv_it=inv.find(i=>i.mat_cod===VC_S.mat_cod);
if(inv_it){inv_it.cons+=tot;inv_it.saldo=Math.max(0,inv_it.saldo-tot);}
C.vcs.unshift({id:vcId,obra_id:OID,fecha:$(‘vcf’).value,encargado:o.encargado,estado:‘confirmado’,
items:[{mat_cod:VC_S.mat_cod,mat_desc:VC_S.mat_cod,guia:VC_S.guia,unidad:VC_S.unidad,total:tot,dists}]});
closeM();render();
toast(`VC confirmado — ${dists.length} quiebre(s) ✓`);
}
